🔍 4. STEP 1 — NORMALIZATION (PURE)



function normalizeStats(stats) {

  return {

    energy: stats.energy / 100,

    focus: stats.focus / 100,

    battery: stats.mentalBattery / 100,

    load: stats.mentalLoad / 100,

    physical: stats.physicalHealth / 100,

    sleep: Math.min(1, stats.sleep / 8)

  };

}

👉 Geen logica hier. Alleen data schoonmaken.









🚫 5. STEP 2 — HARD CONSTRAINTS (KRITISCH)





Hier beslis je wat NIET MAG

function getConstraints(normalized) {



  return {

    blockPhysical: normalized.physical < 0.3,

    blockMental: normalized.battery < 0.3,

    forceRecovery:

      normalized.battery < 0.2 ||

      normalized.energy < 0.2 ||

      normalized.sleep < 0.3

  };



}

👉 Dit is waar 90% van apps falen

🧠 6. STEP 3 — EFFECTIVE STATS (INTERACTIES)





Hier komt nuance:

function getEffectiveStats(n) {



  const effectiveBattery = n.battery - (n.load * 0.5);



  return {

    ...n,

    effectiveBattery: Math.max(0, effectiveBattery)

  };



}









🧬 7. STEP 4 — STATE ENGINE



function getBrainState(e) {



  const score =

    e.energy * 0.2 +

    e.focus * 0.2 +

    e.effectiveBattery * 0.3 +

    e.sleep * 0.15 +

    e.physical * 0.15;



  if (score >= 0.75) return "PEAK";

  if (score >= 0.55) return "OPTIMAL";

  if (score >= 0.35) return "LIMITED";

  if (score >= 0.2) return "DRAINED";

  return "CRASH";

}

👉 score = alleen voor “algemene toestand”

NIET voor beperkingen









🎯 8. STEP 5 — TASK ENGINE (HIER GEBEURT HET)







8.1 Filter op constraints



function filterTasks(tasks, constraints) {



  return tasks.filter(task => {



    if (constraints.forceRecovery && task.type !== "recovery") return false;



    if (constraints.blockPhysical && task.type === "physical") return false;



    if (constraints.blockMental && task.type === "mental") return false;



    return true;

  });



}









8.2 State-based selectie



function selectTasks(tasks, state) {



  const config = {

    PEAK: { count: 5 },

    OPTIMAL: { count: 4 },

    LIMITED: { count: 3 },

    DRAINED: { count: 2 },

    CRASH: { count: 1 }

  };



  return tasks.slice(0, config[state].count);

}









💰 9. XP ENGINE (EINDELIJK CONSISTENT)



function calculateXP(task, state, effectiveStats) {



  const stateMultiplier = {

    PEAK: 1.3,

    OPTIMAL: 1.0,

    LIMITED: 0.85,

    DRAINED: 0.6,

    CRASH: 0.3

  };



  const effortFactor =

    task.type === "mental"

      ? effectiveStats.effectiveBattery

      : effectiveStats.physical;



  return Math.floor(task.baseXP * stateMultiplier[state] * effortFactor);

}









🔁 10. FULL PIPELINE (DIT IS WAT JE CODE MOET DOEN)



function runEngine(userState, allTasks) {



  const normalized = normalizeStats(userState.stats);



  const constraints = getConstraints(normalized);



  const effectiveStats = getEffectiveStats(normalized);



  const state = getBrainState(effectiveStats);



  const availableTasks = filterTasks(allTasks, constraints);



  const selectedTasks = selectTasks(availableTasks, state);



  return {

    state,

    tasks: selectedTasks,

    constraints

  };



}

function classifyTask(name) {



  const lower = name.toLowerCase();



  if (lower.includes("study") || lower.includes("learn"))

    return TASK_PRESETS.study;



  if (lower.includes("clean") || lower.includes("opruimen"))

    return TASK_PRESETS.cleaning;



  if (lower.includes("gym") || lower.includes("workout"))

    return TASK_PRESETS.workout;



  return TASK_PRESETS.relax;

}

⚖️ 6. MATCHING ENGINE (DIT IS WAAR JE APP “SNAPT”)



function isTaskSuitable(task, effectiveStats) {



  const mentalFit =

    effectiveStats.effectiveBattery >= task.mentalLoad * 100;



  const physicalFit =

    effectiveStats.physical >= task.physicalLoad * 100;



  return mentalFit && physicalFit;

}
